1. Backend Configuration

ForgeFlow requires a Node.js backend to securely communicate with the Gemini API. Deploy the following code to your server.

server.js (Node.js/Express)
const express = require('express'); const cors = require('cors'); const helmet = require('helmet'); const { GoogleGenerativeAI } = require('@google/generative-ai'); require('dotenv').config(); const app = express(); app.use(helmet()); app.use(cors({ origin: process.env.FRONTEND_URL })); app.use(express.json({ limit: '1mb' })); const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY); app.post('/api/generate', async (req, res) => { try { const { prompt, industry, style } = req.body; const model = genAI.getGenerativeModel({ model: "gemini-1.5-pro" }); const systemPrompt = `You are an elite senior website designer... Return ONLY a JSON object with: projectName, title, description, html, css, js, seoTitle, seoDescription.`; const result = await model.generateContent([systemPrompt, prompt]); const response = await result.response; res.json(JSON.parse(response.text())); } catch (error) { res.status(500).json({ error: "Generation failed" }); } }); app.listen(process.env.PORT || 3000);
.env.example
GEMINI_API_KEY=your_key_here PORT=3000 FRONTEND_URL=https://your-github-pages-url.com
